// Вода — плавные блики
#define SPEED 0.6
#define WAVE_HEIGHT 0.04
#define WAVE_SCALE 5.0

float rand(vec2 co)
{
    return fract(sin(dot(co, vec2(12.9898, 78.233))) * 43758.5453);
}

float noise(vec2 p)
{
    vec2 i = floor(p);
    vec2 f = fract(p);
    vec2 u = f * f * (3.0 - 2.0 * f);
    return mix(mix(rand(i + vec2(0.0, 0.0)), rand(i + vec2(1.0, 0.0)), u.x),
               mix(rand(i + vec2(0.0, 1.0)), rand(i + vec2(1.0, 1.0)), u.x), u.y);
}

void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    // Нормализация
    vec2 uv = fragCoord / iResolution.xy;
    float t = iTime * SPEED;

    // 1. Волны (искажение воды)
    float wave1 = sin(uv.x * WAVE_SCALE + t) * cos(uv.y * WAVE_SCALE + t * 0.5);
    float wave2 = sin(uv.y * WAVE_SCALE * 1.5 - t * 0.8) * cos(uv.x * WAVE_SCALE * 0.8 + t);
    
    float height = (wave1 + wave2) * WAVE_HEIGHT;
    vec2 distortedUV = uv + vec2(height, height);

    // 2. Цвет воды (бирюзовый)
    vec3 shallowColor = vec3(0.0, 0.6, 0.7); 
    vec3 deepColor = vec3(0.0, 0.1, 0.3);    
    float depth = pow(1.0 - uv.y, 2.0);
    vec3 waterColor = mix(shallowColor, deepColor, depth);

    // 3. БЛИКИ (крупные и плавные)
    // Берем шум с меньшей частотой (20.0 вместо 50.0)
    float glintNoise = noise(distortedUV * 20.0 + t);
    
    // Убираем пики, оставляем плавные пятна
    float glint = smoothstep(0.6, 0.8, glintNoise); 
    
    // Добавляем блики
    waterColor += vec3(1.0, 1.0, 1.0) * glint * 0.5;

    // 4. Финал
    fragColor = vec4(waterColor, 1.0);
}
